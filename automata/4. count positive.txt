%{
#include <stdio.h>
int pos = 0, neg = 0, frac = 0;
%}

%%
[0-9]+\.[0-9]+ { frac++; } /* Matches decimals like 2.4 */
-[0-9]+        { neg++; }  /* Matches negatives like -3 */
[0-9]+         { pos++; }  /* Matches standard positive integers */
\n             { printf("positive numbers=%d\nnegative numbers=%d\nfraction numbers=%d\n", pos, neg, frac); pos=0; neg=0; frac=0; }
[ \t]+         ; /* Ignore spacing between numbers */
.              ;
%%

int yywrap() { return 1; }

int main() {
    printf("Enter numbers:\n");
    yylex();
    return 0;
}
